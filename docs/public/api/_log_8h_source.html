76// 断言（错误处理契约的"逻辑不变量"层）
77//
78// 契约约定：
79// - 致命初始化失败 → return false + ST_CORE_ERROR
80// - 非致命降级 → return nullptr + ST_CORE_WARN
81// - 逻辑不变量（不应发生）→ ST_CORE_ASSERT（Debug 拦截，Release 编译为 no-op）
82//
83// 用于"若此处为假则代码逻辑一定有 bug"的检查，如"System 必须有 scene"。
84// 不要用于用户输入/可恢复错误（那应该走 WARN + fallback）。
85// ════════════════════════════════════════════════════════════
86#ifdef NDEBUG
87 // Release：断言编译为 no-op
88 #define ST_CORE_ASSERT(cond, msg) ((void)0)
89 #define ST_ASSERT(cond, msg) ((void)0)
90#else
91 // Debug：cond 为假 → 记录 CRITICAL 并 abort
92 #define ST_CORE_ASSERT(cond, msg) \
93 do { if (!(cond)) { \
94 if (auto lg = ::Shit::Log::GetCoreLogger()) \
95 lg->critical("ASSERT FAILED: {} -- {}", std::string(#cond), std::string(msg)); \
96 std::abort(); \
97 } } while (0)
98 #define ST_ASSERT(cond, msg) \
99 do { if (!(cond)) { \
