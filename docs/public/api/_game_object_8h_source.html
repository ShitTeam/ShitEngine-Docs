100 template <typename T, typename... Args>
101 T* addComponent(Args&&... args) {
102 static_assert(std::is_base_of_v<Component, T>, "添加的组件必须继承自 Component！");
103
104 auto type_index = std::type_index(typeid(T));
105
106 if (hasComponent<T>()) { // 是否已经存在
107 return getComponent<T>();
108 }
109
110 // 创建组件
111 auto new_component = std::make_unique<T>(std::forward<Args>(args)...);
112 T* new_component_ptr = new_component.get();
113 new_component->setOwner(this);
114
115 // 先插入容器，再触发生命周期回调。若 onAttach/onCreate 内销毁了 owner
116 //（如 removeGameObject 立即清理），对已释放对象写 map 会堆破坏；先入表则
117 // 回调内可以安全地把自己移除，clean() 也能遍历到它。
118 m_components[type_index] = std::unique_ptr<Component>(new_component.release());
119 if (m_scene) m_scene->bumpGeneration(); // 组件结构变更：通知场景代数递增
120
121 // 捕获回调可能用到的数据：回调内可能销毁 owner（this），之后不能再触碰 this
122 Scene* scene = m_scene;
123 std::weak_ptr<void> lifetime = m_lifetime;
124 const std::string goName = m_name;
125
126 new_component_ptr->onCreate(); // onCreate：轻量初始化
127 if (lifetime.expired()) return nullptr; // 回调销毁了 owner → 组件已随 clean() 释放
128
129 // 若已挂载场景则立即执行 onAttach（注册到 System）。
130 // 注册状态由组件自身的 onAttach 维护（基类默认置 true），此处不强制置位，
131 // 以便"组件先加、驱动系统后注册"时能被 System::init 补挂。
132 if (scene && !new_component_ptr->isRegistered()) {
133 new_component_ptr->onAttach();
134 }
135 if (lifetime.expired()) return nullptr; // onAttach 销毁了 owner
136
137 // P20: 组件 UUID 入场景索引（引用字段寻址；幂等，与 setScene 全量 sweep 兼容）
138 // 防御：onCreate/onAttach 回调可能已 removeComponent 移除并析构本组件
139 //（owner 未销毁时 lifetime 检查不生效），先确认组件仍归属本对象再索引。
140 if (scene) {
141 auto it = m_components.find(type_index);
142 if (it != m_components.end() && it->second.get() == new_component_ptr) {
143 scene->indexComponentUuid(new_component_ptr);
144 }
145 }
146
147 ST_CORE_TRACE("GameObject : {} 已添加 组件 {}", goName, typeid(T).name());
148
149 return new_component_ptr;
150 }

196 void removeComponent(const std::type_index &typeIndex) {
197 if (auto it = m_components.find(typeIndex); it != m_components.end()) {
198 // 先从容器取出再调回调，避免回调内再次 removeComponent 造成迭代器失效/重复回调
199 auto comp = std::move(it->second);
200 m_components.erase(it);
201 if (m_scene) m_scene->bumpGeneration(); // 组件结构变更：通知场景代数递增
202 // P20: 组件 UUID 出索引（引用字段随即失效；onDetach 内再 unregister 也幂等）
203 if (m_scene) m_scene->unindexComponentUuid(comp.get());
204 comp->onDetach();
205 comp->onDestroy();
206 }
207 }
