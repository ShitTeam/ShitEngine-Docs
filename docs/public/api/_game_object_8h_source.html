129 template <typename T, typename... Args>
130 T* addComponent(Args&&... args) {
131 static_assert(std::is_base_of_v<Component, T>, "添加的组件必须继承自 Component！");
132
133 auto type_index = std::type_index(typeid(T));
134
135 if (hasComponent<T>()) { // 是否已经存在
136 return getComponent<T>();
137 }
138
139 // 创建组件
140 auto new_component = std::make_unique<T>(std::forward<Args>(args)...);
141 T* new_component_ptr = new_component.get();
142 new_component->setOwner(this);
143
144 // 先插入容器，再触发生命周期回调。若 onAttach/onCreate 内销毁了 owner
145 //（如 removeGameObject 立即清理），对已释放对象写 map 会堆破坏；先入表则
146 // 回调内可以安全地把自己移除，clean() 也能遍历到它。
147 m_components[type_index] = std::unique_ptr<Component>(new_component.release());
148 if (m_scene) m_scene->bumpGeneration(); // 组件结构变更：通知场景代数递增
149
150 // 捕获回调可能用到的数据：回调内可能销毁 owner（this），之后不能再触碰 this
151 Scene* scene = m_scene;
152 std::weak_ptr<void> lifetime = m_lifetime;
153 const std::string goName = m_name;
154
155new_component_ptr->onCreate(); // onCreate：轻量初始化
156 if (lifetime.expired()) return nullptr; // 回调销毁了 owner → 组件已随 clean() 释放
157
158 // onCreate 回调可能 removeComponent 移除了本组件（owner 存活时 lifetime 检查不生效），
159 // 先确认组件仍归属本对象，再 isRegistered/onAttach，避免解引用已析构实例
160 {
161 auto it = m_components.find(type_index);
162 if (it == m_components.end() || it->second.get() != new_component_ptr) return nullptr;
163 }
164
165 // 若已挂载场景则立即执行 onAttach（注册到 System）。
166 // 注册状态由组件自身的 onAttach 维护（基类默认置 true），此处不强制置位，
167 // 以便"组件先加、驱动系统后注册"时能被 System::init 补挂。
168 if (scene && !new_component_ptr->isRegistered()) {
169 new_component_ptr->onAttach();
170 }
171 if (lifetime.expired()) return nullptr; // onAttach 销毁了 owner
172
173 // P20: 组件 UUID 入场景索引（引用字段寻址；幂等，与 setScene 全量 sweep 兼容）
174 // 防御：onCreate/onAttach 回调可能已 removeComponent 移除并析构本组件
175 //（owner 未销毁时 lifetime 检查不生效），先确认组件仍归属本对象再索引。
176 if (scene) {
177 auto it = m_components.find(type_index);
178 if (it != m_components.end() && it->second.get() == new_component_ptr) {
179 scene->indexComponentUuid(new_component_ptr);
180 }
181 }
182
183 ST_CORE_TRACE("GameObject : {} 已添加 组件 {}", goName, typeid(T).name());
184
185 return new_component_ptr;
186 }

232 void removeComponent(const std::type_index &typeIndex) {
233 if (auto it = m_components.find(typeIndex); it != m_components.end()) {
234 // 先从容器取出再调回调，避免回调内再次 removeComponent 造成迭代器失效/重复回调
235 auto comp = std::move(it->second);
236 m_components.erase(it);
237 if (m_scene) m_scene->bumpGeneration(); // 组件结构变更：通知场景代数递增
238 // P20: 组件 UUID 出索引（引用字段随即失效；onDetach 内再 unregister 也幂等）
239 if (m_scene) m_scene->unindexComponentUuid(comp.get());
240 comp->onDetach();
241 comp->onDestroy();
242 }
243 }
