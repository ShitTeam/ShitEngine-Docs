170 template <typename Res>
171 class TypedResourceCache final : public TypedCacheBase {
172 public:
173 using KeyType = typename ResourceTraits<Res>::KeyType;
174 using HashType = typename ResourceTraits<Res>::HashType;
175
176 explicit TypedResourceCache(ResourceManager& owner) : m_owner(owner) {}
177
179 Res* getOrLoad(const KeyType& key) {
180 if (auto* hit = find(key)) return hit;
181 ST_CORE_DEBUG("{} {} 未命中缓存，正在加载 ...", ResourceTraits<Res>::kTypeName, logKey(key));
182 std::unique_ptr<Res> res = ResourceTraits<Res>::create(m_owner, key);
183 if (!res || !res->load())
184 return nullptr; // 失败不缓存：下次请求重试
185 Res* ptr = res.get();
186 m_cache.emplace(key, std::move(res));
187 return ptr;
188 }
189
190 Res* find(const KeyType& key) const {
191 if (auto it = m_cache.find(key); it != m_cache.end())
192 return it->second.get();
193 return nullptr;
194 }
195
196 bool unload(const KeyType& key) { return m_cache.erase(key) > 0; }
197
198 void clear() override { m_cache.clear(); }
199 size_t size() const override { return m_cache.size(); }
200 std::string_view typeName() const override { return ResourceTraits<Res>::kTypeName; }
201 void forEachResource(const std::function<void(Resource*)>& fn) const override {
202 for (const auto& [key, res] : m_cache) fn(res.get());
203 }
204
205 private:
206 // 日志键显示：字符串键原样，复合键取路径部分
207 static const std::string& logKey(const std::string& k) { return k; }
208 static const std::string& logKey(const FontKey& k) { return k.path; }
209
210 ResourceManager& m_owner;
211 std::unordered_map<KeyType, std::unique_ptr<Res>, HashType> m_cache;
212 };
