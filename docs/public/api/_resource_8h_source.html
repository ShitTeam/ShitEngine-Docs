126 template <typename Key, typename Res, typename Hash = std::hash<Key>>
127 class ResourceCache {
128 public:
130 template <typename Make>
131 Res* getOrLoad(const Key& key, Make&& make) {
132 if (auto it = m_cache.find(key); it != m_cache.end())
133 return it->second.get();
134 std::unique_ptr<Res> res = make();
135 if (!res || !res->load())
136 return nullptr;
137 Res* ptr = res.get();
138 m_cache.emplace(key, std::move(res));
139 return ptr;
140 }
141
142 Res* find(const Key& key) const {
143 if (auto it = m_cache.find(key); it != m_cache.end())
144 return it->second.get();
145 return nullptr;
146 }
147
148 bool unload(const Key& key) { return m_cache.erase(key) > 0; }
149 void clear() { m_cache.clear(); }
150 bool empty() const { return m_cache.empty(); }
151 size_t size() const { return m_cache.size(); }
152
153 const std::unordered_map<Key, std::unique_ptr<Res>, Hash>& entries() const { return m_cache; }
154
155 private:
