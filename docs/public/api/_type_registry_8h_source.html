73inline std::string DemangleTypeName(const char* mangled) {
74#if defined(SHIT_HAS_CXXABI)
75 int status = 0;
76 char* demangled = abi::__cxa_demangle(mangled, nullptr, nullptr, &status);
77 if (status == 0 && demangled) {
78 std::string result(demangled);
79 ::free(demangled);

81 }
82#endif
