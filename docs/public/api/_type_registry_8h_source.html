88// GCC/MinGW: abi::__cxa_demangle, MSVC: typeid 直接可读
89inline std::string DemangleTypeName(const char* mangled) {
90#if defined(SHIT_HAS_CXXABI)
91 int status = 0;
92 char* demangled = abi::__cxa_demangle(mangled, nullptr, nullptr, &status);
93 if (status == 0 && demangled) {
94 std::string result(demangled);
95 ::free(demangled);
96 return result;
97 }
98#endif
99 // MSVC 或 demangle 失败：直接返回原名
100 return std::string(mangled);
